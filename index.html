🔐 Firestore Security Rules
Copy these rules into Firebase Console → Firestore → Rules → Publish
rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // Users — anyone can read (needed for PIN login) // Only existing admin can write (role changes) match /users/{userId} { allow read: if true; allow write: if true; } // Settings — anyone can read; only admin writes match /settings/{doc} { allow read: if true; allow write: if true; } // Transactions — admin & member: full CRUD // child/other: create + read only match /transactions/{txId} { allow read, create: if true; allow update, delete: if true; } // Goals — admin & member: full CRUD // child/other: read only match /goals/{goalId} { allow read, create, update: if true; allow delete: if true; } // Bank Accounts — admin only: write // all members: read match /bankAccounts/{bankId} { allow read: if true; allow write: if true; } } } // NOTE: Full server-side role enforcement requires Firebase Authentication. // Current rules allow all (PIN auth is client-side). UI enforces admin/member/child roles.
PIN-based login: DB-level security uses "allow all". Admin enforcement is in the app UI. For production, integrate Firebase Authentication.
